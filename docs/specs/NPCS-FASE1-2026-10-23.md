## 4. Código de referência: geração determinística dos perfis

Criar `tools/generate-npc-profiles.mjs` com o conteúdo abaixo. Esse arquivo apenas combina os **dados existentes**, sem produzir novas características ou efeitos. Executar após copiar os dois JSONs canônicos. O código gerado em `data/continuous/npc-profiles.js` pode ser versionado.

```js
// tools/generate-npc-profiles.mjs
import fs from 'node:fs';
import path from 'node:path';

const catalog = JSON.parse(fs.readFileSync('data/npcs/catalogo-v1.json', 'utf8'));
const manifest = JSON.parse(fs.readFileSync('data/continuous/npc-art-manifest.json', 'utf8'));
const details = new Map(catalog.npcs.map(p => [p.id, p]));
const images = manifest.npcs;
if (!Array.isArray(images) || images.length !== 16) throw Error('Expected 16 NPC images');
if (details.size !== 16) throw Error('Expected 16 canonical NPC profiles');

const profile = Object.fromEntries(images.map((a) => {
  const d = details.get(a.id);
  if (!d || a.name !== d.nome || a.type !== d.tipo)
    throw Error(`Identity mismatch for ${a.id}`);
  if (!/^assets\/npcs\/npc-\d{2}-[a-z0-9-]+\.png$/.test(a.canonical))
    throw Error(`Invalid canonical image path: ${a.canonical}`);
  return [a.id, {
    id: a.id,
    name: d.nome,
    type: d.tipo,
    archetype: d.arquetipo,
    group: d.grupo,
    portrait: '/' + a.canonical,
    age: d.idade ?? null,
    roleLabel: d.papel ?? null,
    summary: d.influencia ?? null,
    strengths: d.forcas ?? [],
    watchouts: d.pontos_de_atencao ?? [],
    favorableConditions: d.condicoes_favoraveis ?? [],
    adverseConditions: d.condicoes_adversas ?? [],
    situationalSignals: d.sinais_situacionais ?? [],
    proposedSynergies: d.sinergias_propostas ?? [],
    environments: d.ambientes ?? null,
    proposedEffects: d.pontos ?? null,
    status: d.status ?? 'em-revisao',
    disclaimer: 'Perfil narrativo fictício; não determina saúde, doença ou aptidão.'
  }];
}));

fs.mkdirSync(path.dirname('data/continuous/npc-profiles.js'), {recursive:true});
fs.writeFileSync('data/continuous/npc-profiles.js',
  '/* Gerado a partir dos catálogos canônicos; não editar à mão. */\n' +
  'export const NPC_PROFILES = Object.freeze(' + JSON.stringify(profile, null, 2) + ');\n' +
  'export const npcProfile = id => NPC_PROFILES[id] || null;\n', 'utf8');
console.log('Perfis NPC gerados:', Object.keys(profile).length);
```

const narrativeList = (values) => Array.isArray(values) && values.length
  ? `<ul>${values.map(v => `<li>${esc(v.replaceAll('_', ' '))}</li>`).join('')}</ul>`

function narrative(p) {

## 8. Testes automatizados sugeridos

Criar `tests/npc-profiles.test.mjs` usando `node:test`. O teste abaixo pressupõe que o gerador já foi executado **e os 16 PNGs estão incluídos no commit**. Se ainda faltar imagem, usar fallback durante desenvolvimento, mas **não declarar a entrega final concluída**.

```js
// tests/npc-profiles.test.mjs
import test from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';
import {EMPLOYEES} from '../data/continuous/employees.js';
import {NPC_PROFILES} from '../data/continuous/npc-profiles.js';
import {fresh, staffView} from '../simulation/company-living.js';

const ids = EMPLOYEES.map(p => p.id);

test('16 NPCs preservam IDs, arte única e fonte válida', () => {
  assert.equal(ids.length, 16);
  assert.equal(new Set(ids).size, 16);
  assert.equal(Object.keys(NPC_PROFILES).length, 16);
  const paths = EMPLOYEES.map(p => {
    assert.ok(p.profile, `falta perfil ${p.id}`);
    assert.equal(p.name, p.profile.name);
    assert.equal(p.type, p.profile.type);
    assert.ok(p.portrait.startsWith('/assets/npcs/npc-'));
    const local = p.portrait.slice(1);
    assert.ok(fs.existsSync(local), `falta asset ${local}`);
    return local;
  });
  assert.equal(new Set(paths).size, 16);
});

test('arte e personalidade não modificam saúde simulada', () => {
  const before = staffView(fresh()).map(({id,health}) => [id, health]);
  for (const p of EMPLOYEES) assert.ok(p.profile.disclaimer.includes('não determina saúde'));
  const after = staffView(fresh()).map(({id,health}) => [id, health]);
  assert.deepEqual(after, before);
});

test('dados ausentes permanecem ausentes, sem inventar idades', () => {
  for (const p of EMPLOYEES.filter(p => Number(p.id.slice(-2)) >= 9)) {
    assert.equal(p.age, null);
    assert.equal(p.profile.age, null);
  }
});
```
